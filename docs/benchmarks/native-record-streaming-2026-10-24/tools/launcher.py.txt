#!/usr/bin/python3 -I
"""Strict local launcher for the reviewed fixed two-arm adapter experiment; no execution on import."""
import json
import os
import pathlib
import re
import stat
import sys

REAL = "/usr/bin/systemd-run"

def main():
    # Controller creates this owned mode0600 immutable config before any arm.
    config_path = os.environ["OOJURO_BASELINE_CONTROL"]
    fd = os.open(config_path, os.O_RDONLY | os.O_NOFOLLOW)
    with os.fdopen(fd, "rb") as stream:
        info = os.fstat(stream.fileno())
        assert stat.S_ISREG(info.st_mode) and info.st_uid == os.getuid()
        assert stat.S_IMODE(info.st_mode) == 0o600 and info.st_size <= 16384
        config = json.load(stream)
    assert config["format"] == 1
    parent = os.getppid()
    assert os.readlink(f"/proc/{parent}/exe") == config["driver"]
    args = sys.argv[1:]
    assert len(args) == 12 and args[:4] == ["--user", "--scope", "--quiet", "--collect"]
    assert args[5:8] == ["--property=MemoryMax=3G", "--property=MemorySwapMax=0", "--"]
    match = re.fullmatch(r"--unit=(oojuro-arm-(native_before|native_after)-"
                         + str(parent) + r"-[0-9]+)", args[4])
    assert match is not None
    unit, label = match.groups()
    assert args[8:] == config["arms"][label]
    assert set(config["arms"]) == {"native_before", "native_after"}
    assert all(v[1:3] == ["serve", "--arm"] and len(v) == 4 for v in config["arms"].values())
    # Preserve the existing harness name. The append-only exact-name inventory
    # exists BEFORE systemd receives the request, including uncertain launches.
    start_ticks = pathlib.Path("/proc/self/stat").read_text().rsplit(")", 1)[1].split()[19]
    record = json.dumps({"unit": unit + ".scope", "launcher_pid": os.getpid(), "start_ticks": start_ticks,
                         "harness_pid": parent, "label": label},
                        separators=(",", ":")).encode() + b"\n"
    fd = os.open(config["inventory"], os.O_WRONLY | os.O_APPEND | os.O_NOFOLLOW)
    try:
        info = os.fstat(fd)
        assert stat.S_ISREG(info.st_mode) and info.st_uid == os.getuid()
        assert stat.S_IMODE(info.st_mode) == 0o600
        assert os.write(fd, record) == len(record)
        os.fsync(fd)
    finally:
        os.close(fd)
    # One reviewed true smoke must establish these installed scope properties.
    # No fallback without the finite lifetime.
    os.execv(REAL, [REAL, *args[:7], "--property=RuntimeMaxSec=300", *args[7:]])

if __name__ == "__main__":
    main()
