def sha(p):
    with pathlib.Path(p).open("rb") as f:return hashlib.file_digest(f,"sha256").hexdigest()
def save(p,value):
    tmp=p.with_name(p.name+".pending")
    fd=os.open(tmp,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    with os.fdopen(fd,"w") as f:json.dump(value,f,indent=2);f.write("\n");f.flush();os.fsync(f.fileno())
    os.replace(tmp,p)
    fd=os.open(p.parent,os.O_RDONLY|os.O_DIRECTORY)
    try:os.fsync(fd)
    finally:os.close(fd)
def command(run,phase,n=0):
    common=["--runs","1","--mem-max","3G","--data",str(DATA),"--keep"]
    if phase=="gates":
        args=[str(C),"gates",*common,"--work",str(run/"gates-stores"),"--out",str(run/"gates.json")]
        order=LABELS

              "--thinking","index","--k","20","--pool","20","--batch","1000","--work",str(run/f"stores/round-{n}"),
              "--out",str(run/f"round-{n}.json"),"--dump-pools",str(run/f"pools-round-{n}")]

    for label in order:args+=["--arm",SPECS[label]]

def reconcile(run):
    # Scope placement is observed, never inferred from the app.slice convention.
    own=pathlib.Path('/proc/self/cgroup').read_text().strip().split('0::',1)[1]
    assert '/app.slice/' in own
    manager=pathlib.Path('/sys/fs/cgroup')/own.split('/app.slice/',1)[0].lstrip('/')
    unknown=[];result=[];end=time.monotonic()+100
    def issue(phase,index,error):
        unknown.append({'phase':phase,'row':index,'class':type(error).__name__})
    try:lines=(run/'scopes.jsonl').read_text().splitlines()
    except Exception as error:
        issue('inventory-read',None,error);lines=[]
    if len(lines)>256:
        issue('inventory-cap',None,ValueError());lines=lines[:256]
    observations={}
    pathfile=run/'scope-observations.json'
    if pathfile.exists():
        try:
            observations=json.loads(pathfile.read_text())
            assert isinstance(observations,dict)
        except Exception as error:issue('observations-read',None,error);observations={}
    def checked_path(unit,value):
        assert isinstance(value,str) and value.startswith('/') and '..' not in pathlib.PurePosixPath(value).parts
        path=pathlib.Path('/sys/fs/cgroup')/value.lstrip('/')
        assert path.is_relative_to(manager) and path.name==unit
        return path
    def capture(unit,stdout,index):
        reported=[line.split('=',1)[1] for line in stdout.splitlines() if line.startswith('ControlGroup=')]
        assert len(reported)<=1
        previous=observations.get(unit,[])
        assert isinstance(previous,list)
        for value in previous:checked_path(unit,value)
        if reported and reported[0]:
            checked_path(unit,reported[0])
            if reported[0] not in previous:
                observations[unit]=previous+[reported[0]]
                # Save the actual path BEFORE any termination command.
                try:save(pathfile,observations)
                except Exception as error:issue('observations-save',index,error)
        return observations.get(unit,[])
    def query(args):
        left=end-time.monotonic();assert left>0
        return subprocess.run(args,capture_output=True,text=True,timeout=min(5,left))
    def named_paths(unit):
        found=[];count=0
        def failed(error):raise error
        for parent,dirs,files in os.walk(manager,onerror=failed):
            count+=1;assert count<=4096 and time.monotonic()<end
            if unit in dirs:found.append(str(pathlib.Path(parent)/unit))
        return found
    seen=set()
    for index,line in enumerate(lines):
        entry={'row':index}
        try:
            r=json.loads(line)
            assert type(r['harness_pid']) is int and r['harness_pid']>0
            assert type(r['launcher_pid']) is int and r['launcher_pid']>0
            assert isinstance(r['start_ticks'],str) and r['start_ticks'].isdigit()
            unit=r['unit'];assert re.fullmatch(r'oojuro-arm-(native_current|native_control|sqlite|jsonl_zstd)-'+str(r['harness_pid'])+r'-[0-9]+\.scope',unit)
            assert unit not in seen;seen.add(unit);entry['unit']=unit
            q=['/usr/bin/systemctl','--user','show','--property=LoadState','--property=ActiveState','--property=ControlGroup',unit]
            state=query(q)
            actual=capture(unit,state.stdout,index)
            entry['observed_control_groups']=list(actual)
            if any('ActiveState='+v in state.stdout for v in ['active','activating','deactivating','failed']):
                for cmd in [['kill','--signal=KILL','--kill-whom=all',unit],['stop',unit]]:
                    try:
                        termination=query(['/usr/bin/systemctl','--user',*cmd])
                        entry.setdefault('termination_exits',[]).append(termination.returncode)
                    except Exception as error:issue('unit-terminate',index,error)
            until=min(end,time.monotonic()+5)
            while True:
                state=query(q);actual=capture(unit,state.stdout,index)
                missing=state.returncode!=0 and 'not found' in state.stderr.lower()
                dead=missing or 'LoadState=not-found' in state.stdout or 'ActiveState=inactive' in state.stdout
                actual_absent=all(not checked_path(unit,value).exists() for value in actual)
                # Already collected units may have no remaining reported path.
                # Search the ACTUAL manager hierarchy for the exact unit basename.
                remaining_named=named_paths(unit) if dead else []
                pid=pathlib.Path('/proc')/str(r['launcher_pid'])/'stat'
                try:pid_same=pid.read_text().rsplit(')',1)[1].split()[19]==r['start_ticks']
                except FileNotFoundError:pid_same=False
                if dead and actual_absent and not remaining_named and not pid_same:break
                assert time.monotonic()<until
                time.sleep(.05)
            entry.update(state=state.stdout,query_exit=state.returncode,observed_control_groups=list(actual),
                         observed_paths_absent=actual_absent,manager_hierarchy=str(manager),
                         exact_unit_named_paths_remaining=remaining_named,exact_original_pid_absent=True,
                         cgroup_proof='observed paths plus exact-name hierarchy search' if actual else 'already-collected; exact-name hierarchy search only',
                         status='reconciled')
        except Exception as error:
            issue('unit-reconcile',index,error);entry['status']='unknown'
        result.append(entry)
    return {'rows':result,'unknown':unknown,'manager_hierarchy':str(manager)}

def validate_card(path,phase):
    card=json.loads(path.read_text());assert card["schema"]=="oojuro-scorecard-v1"
    assert {r["label"] for r in card["arms"]}==set(LABELS) and len(card["arms"])==4
    assert card["settings"]["runs"]==1
    for row in card["arms"]:
        assert row["error"] is None and row["gates"] and all(g["pass"] for g in row["gates"].values())
        assert row["info"]["commit"]==(CONTROL if row["label"]=="native_control" else CURRENT)[:12]
        if phase=="gates":
            assert set(row["gates"])=={"byte_exact","byte_exact_after_reopen","get_range_order","as_of_visibility","freshness","determinism","crash_recovery"}
        else:
            assert set(row["gates"])=={"no_future_leak","byte_exact","freshness"}
            for section,key in [("memory","peak_rss_bytes"),("memory","peak_cgroup_bytes"),("speed","returned_original_bytes")]:
                assert row[section][key]["n"]==1 and row[section][key]["median"]>0
    if phase!="gates":
        s=card["settings"]
        assert all(s[k]==v for k,v in {"steps":201,"trajectories":10,"background_target_records":20000,"thinking":"index","k":20,"pool":20}.items())
        assert len({r["speed"]["store_records_end"]["median"] for r in card["arms"]})==1
    return card

def inventory(path):
    apparent=allocated=files=0
    for p in sorted(path.rglob("*")):
        info=p.lstat();assert stat.S_ISREG(info.st_mode) or stat.S_ISDIR(info.st_mode)
        if stat.S_ISREG(info.st_mode):
            apparent+=info.st_size;allocated+=info.st_blocks*512;files+=1
    assert files>0
    return {"files":files,"apparent_bytes":apparent,"allocated_bytes":allocated}
def main():

    assert sha(DATA/"prepared/turnloop/swe-agent.jsonl")==INPUT_SHA

    else:

        if a.phase=="workload":
