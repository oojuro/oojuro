#!/bin/sh
set -eu
export ZIG_GLOBAL_CACHE_DIR=<HOME>/.sa-agents/oojuro-toolchain/zig-cache
export ZIG_LOCAL_CACHE_DIR=<HOME>/.sa-agents/oojuro-toolchain/zig-cache
for arg in "$@"; do
 case "$arg" in --target=*|-Wl,--no-undefined-version|-march=native|-mtune=native) ;; *) set -- "$@" "$arg" ;; esac
 shift
done
exec <HOME>/.sa-agents/oojuro-toolchain/zig/zig cc -target x86_64-linux-gnu.2.39 -mcpu=baseline "$@"
